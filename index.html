Run this in your Supabase SQL Editor if table is not created yet:

CREATE TABLE IF NOT EXISTS feedback_records (
  id BIGSERIAL PRIMARY KEY,
  customer_id TEXT,
  customer_name TEXT,
  contact_no TEXT,
  calling_date TEXT,
  feedback TEXT,
  desp_code TEXT,
  portfolio TEXT,
  manager_name TEXT,
  caller_name TEXT,
  source_file TEXT,
  uploaded_by TEXT,
  is_duplicate BOOLEAN DEFAULT FALSE,
  duplicate_tag TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
